  def offgrid_cl(axis, pitch_dbu, offset_dbu, base_dbu = nil)
    new_data = RBA::Region::new
    self.data.each do |p|
      bb = p.bbox
      if base_dbu
        if axis == :y
          next if bb.bottom % base_dbu != 0 || bb.top % base_dbu != 0
        else
          next if bb.left % base_dbu != 0 || bb.right % base_dbu != 0
        end
      end
      cl = axis == :y ? (bb.bottom + bb.top) / 2 : (bb.left + bb.right) / 2
      if (cl - offset_dbu) % pitch_dbu != 0
        if axis == :y
          new_data.insert(p.moved(0, cl - bb.bottom))
        else
          new_data.insert(p.moved(cl - bb.left, 0))
        end
      end
    end
    DRC::DRCLayer::new(@engine, new_data)
  end

  def ongrid_ep(gx_dbu, gy_dbu)
    result = RBA::EdgePairs::new
    self.data.each do |poly|
      pts = poly.each_point_hull.to_a
      pts.each_with_index do |pt, idx|
        offgrid_x = (gx_dbu > 1) && (pt.x % gx_dbu != 0)
        offgrid_y = (gy_dbu > 1) && (pt.y % gy_dbu != 0)
        if offgrid_x || offgrid_y
          pp = pts[(idx - 1) % pts.size]
          np = pts[(idx + 1) % pts.size]
          e1 = RBA::Edge::new(pp, pt)
          e2 = RBA::Edge::new(pt, np)
          result.insert(RBA::EdgePair::new(e1, e2))
        end
      end
    end
    DRC::DRCLayer::new(@engine, result)
  end

## Channel and Batching Strategy

The cu_pool channel (trial:i01.cu.def:VIA_VIA56_2_2_66_58.01) targets individual via cell definitions and applies repairs directly; its ops are recorded as applied and are excluded from subsequent unit_gate assembly passes (evidenced by assemble_drops in trial:i02.ug.whole_design.00). The unit_gate channel (trial:i02.ug.whole_design.00, trial:i04.ug.whole_design.00) operates on whole-design sweeps and gates in batches based on connectivity preservation, tolerating cross-layer side effects such as new lower-metal violations if conn_preserved is true; a batch that introduces zero new violations is also accepted. When cu_pool has already committed via-cell resizes, the unit_gate assembler drops those ops automatically; design the two channels' repair scopes to be non-overlapping to avoid redundant drop tracking (trial:i01.cu.def:VIA_VIA56_2_2_66_58.01, trial:i02.ug.whole_design.00).
