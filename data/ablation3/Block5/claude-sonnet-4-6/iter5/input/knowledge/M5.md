  def offgrid_cl(axis, pitch_dbu, offset_dbu, base_dbu = nil)
    new_data = RBA::Region::new
    self.data.each do |p|
      bb = p.bbox
      if base_dbu
        if axis == :y
          next if bb.bottom % base_dbu != 0 || bb.top % base_dbu != 0
        else
          next if bb.left % base_dbu != 0 || bb.right % base_dbu != 0
        end
      end
      cl = axis == :y ? (bb.bottom + bb.top) / 2 : (bb.left + bb.right) / 2
      if (cl - offset_dbu) % pitch_dbu != 0
        if axis == :y
          new_data.insert(p.moved(0, cl - bb.bottom))
        else
          new_data.insert(p.moved(cl - bb.left, 0))
        end
      end
    end
    DRC::DRCLayer::new(@engine, new_data)
  end

  def ongrid_ep(gx_dbu, gy_dbu)
    result = RBA::EdgePairs::new
    self.data.each do |poly|
      pts = poly.each_point_hull.to_a
      pts.each_with_index do |pt, idx|
        offgrid_x = (gx_dbu > 1) && (pt.x % gx_dbu != 0)
        offgrid_y = (gy_dbu > 1) && (pt.y % gy_dbu != 0)
        if offgrid_x || offgrid_y
          pp = pts[(idx - 1) % pts.size]
          np = pts[(idx + 1) % pts.size]
          e1 = RBA::Edge::new(pp, pt)
          e2 = RBA::Edge::new(pt, np)
          result.insert(RBA::EdgePair::new(e1, e2))
        end
      end
    end
    DRC::DRCLayer::new(@engine, result)
  end

**Rule Taxonomy**

M5 geometry is governed by independent horizontal (x-axis) and vertical (y-axis) rule sets plus auxiliary grid and topology rules.

Horizontal width: M5.W.1 prohibits horizontal widths below 24 nm. M5.W.2 prohibits horizontal widths exceeding 480 nm. M5.W.3 prohibits widths that are exact even-integer multiples of 24 nm: 48, 96, 144, 192, 240, 288, 336, 384, 432, or 480 nm. M5.W.4 prohibits widths of 72, 168, 264, 360, or 456 nm, which cause a polygon to span an even number of minimum-width routing tracks. M5.S.1 sets minimum horizontal spacing between any two M5 polygon edges at 24 nm regardless of edge length or mask color.

Vertical width: M5.W.5 sets minimum vertical width at 44 nm. M5.S.2 sets minimum vertical spacing at 40 nm. Grid and topology: M5.AUX.1 requires every M5 vertical edge to lie on a 24 nm x-grid (checked via ongrid with gx=24 nm, gy=1 dbu).
